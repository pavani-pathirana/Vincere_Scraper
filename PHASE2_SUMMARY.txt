from core import BaseScraper, ArticleMetadata, Article
from bs4 import BeautifulSoup

class AdaDeranaScraper(BaseScraper):
    def get_category_urls(self):
        return {
            'politics': 'https://sinhala.adaderana.lk/politics',
            'sports': 'https://sinhala.adaderana.lk/sports'
        }
    
    def parse_article_list_page(self, html, category):
        soup = BeautifulSoup(html, 'lxml')
        articles = []
        for item in soup.select('.news-item'):
            articles.append(ArticleMetadata(
                url=item.select_one('a')['href'],
                headline=item.select_one('.headline').get_text(),
                category=category
            ))
        return articles
    
    def parse_article_page(self, html, metadata):
        soup = BeautifulSoup(html, 'lxml')
        
        headline = soup.select_one('.article-headline').get_text()
        body_html = soup.select_one('.article-body')
        body = self.text_processor.process(str(body_html))
        
        return Article(
            headline=headline,
            body=body,
            source=self.website_name,
            category=metadata.category,
            date=soup.select_one('.date').get_text(),
            url=metadata.url,
            extraction_timestamp=datetime.now().isoformat()
        )
    
    # ... implement other abstract methods ...
